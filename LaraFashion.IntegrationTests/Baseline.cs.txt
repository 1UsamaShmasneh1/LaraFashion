using LaraFashion.Data;
using LaraFashion.Models;
using LaraFashion.Services;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.Logging.Abstractions;

var root = Path.Combine(Path.GetTempPath(), "LaraFashionTests", Guid.NewGuid().ToString("N"));
Directory.CreateDirectory(root);
var paths = new AppStoragePaths(Path.Combine(root, "test.db"), Path.Combine(root, "uploads"));
Directory.CreateDirectory(paths.UploadsPath);
var options = new DbContextOptionsBuilder<AppDbContext>().UseSqlite($"Data Source={paths.DatabasePath};Pooling=False").Options;
AppDbContext Db() => new(options);
var config = new ConfigurationBuilder().AddInMemoryCollection(new Dictionary<string, string?>
{
    ["JwtSettings:SecretKey"] = "integration-test-only-key-012345678901234567890",
    ["JwtSettings:Issuer"] = "tests", ["JwtSettings:Audience"] = "tests", ["JwtSettings:ExpiryMinutes"] = "60"
}).Build();
AdminAuthService Auth(AppDbContext db) => new(db, new JwtService(config), new PasswordHasherService());
OrderService Orders(AppDbContext db) => new(db, Auth(db), NullLogger<OrderService>.Instance, paths);
async Task<Product> Product(params int[] quantities)
{
    await using var db = Db();
    var p = new Product { Id = Guid.NewGuid(), Name = "Test", OriginalPrice = 10,
        Sizes = quantities.Select((q, i) => new ProductSize { Id = Guid.NewGuid(), SizeName = new[] { "S", "M", "L" }[i], Quantity = q }).ToList() };
    db.Products.Add(p); await db.SaveChangesAsync(); return p;
}
CartItem Item(Product p, string size = "S", int q = 1) => new() { ProductId = p.Id, ProductName = "Test", Size = size, Quantity = q, UnitPrice = 10 };
CustomerInfo Customer() => new() { FullName = "Test", PhoneNumber = "0501234567", City = "Test" };
CartDiscountResult Total(int q = 1) => new() { OriginalTotal = 10 * q };
await using (var db = Db()) { await db.Database.MigrateAsync(); await new DatabaseSeeder(db, new PasswordHasherService()).SeedAsync(); }
var reproduced = 0;
var p = await Product(0, 3, 4);
await using (var stale = Db())
{
    await stale.Products.Include(x => x.Sizes).LoadAsync();
    await using (var other = Db()) await other.ProductSizes.Where(x => x.ProductId == p.Id).ExecuteUpdateAsync(s => s.SetProperty(x => x.Quantity, 0));
    // Track zero stock, then another circuit restores a different size.
    stale.ChangeTracker.Clear(); await stale.Products.Include(x => x.Sizes).LoadAsync();
    await using (var other = Db()) await other.ProductSizes.Where(x => x.ProductId == p.Id && x.SizeName == "M").ExecuteUpdateAsync(s => s.SetProperty(x => x.Quantity, 3));
    await new ProductService(stale, paths).CleanDeletableProductsAsync();
    await using var check = Db();
    if (!await check.Products.AnyAsync(x => x.Id == p.Id)) { reproduced++; Console.WriteLine("REPRODUCED: cleanup deletes restocked product from stale tracker"); }
}
p = await Product(2);
await using (var db = Db())
{
    await Orders(db).CreateOrderAsync(Customer(), [Item(p, q: 2), Item(p, q: 2)], Total(4));
    await using var check = Db();
    if (await check.ProductSizes.Where(x => x.ProductId == p.Id).Select(x => x.Quantity).SingleAsync() == -2)
    { reproduced++; Console.WriteLine("REPRODUCED: repeated cart lines overdraw stock"); }
}
p = await Product(1);
await using (var first = Db()) await using (var stale = Db())
{
    await stale.ProductSizes.Where(x => x.ProductId == p.Id).LoadAsync();
    await Orders(first).CreateOrderAsync(Customer(), [Item(p)], Total());
    await Orders(stale).CreateOrderAsync(Customer(), [Item(p)], Total());
    await using var check = Db();
    if (await check.OrderItems.CountAsync(x => x.ProductId == p.Id) == 2)
    { reproduced++; Console.WriteLine("REPRODUCED: stale second circuit sells last piece twice"); }
}
p = await Product(20);
var numbers = new HashSet<string>();
for (var i = 0; i < 10; i++)
{
    await using var db = Db();
    var o = await Orders(db).CreateOrderAsync(Customer(), [Item(p)], Total());
    if (!numbers.Add(o.OrderNumber)) { reproduced++; Console.WriteLine("REPRODUCED: different GUID orders share timestamp display number"); break; }
}
Console.WriteLine($"Baseline defects reproduced: {reproduced}/4; isolated SQLite: {paths.DatabasePath}");
return reproduced == 4 ? 0 : 1;
